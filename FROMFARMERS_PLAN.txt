# FromFarmers App Implementation Plan

A comprehensive step-by-step guide to create the **FromFarmers** app, connecting farmers directly with buyers. This plan exports the architecture and design patterns from the **stylrSA/hairprosdirectory** codebase.

---

## 📋 Project Overview

**FromFarmers** is a marketplace platform that:
- Connects farmers (sellers) with regular consumers (buyers)
- Allows farmers to list their agricultural products
- Provides farmer profile pages with all their listed products
- Has detailed product pages with full information
- Is fully responsive for mobile and desktop

### Phase 1: Project Setup

#### Step 1.1: Initialize Next.js Project
```bash
# Navigate to parent directory
cd c:\Users\ramos\all_coding

# Create new Next.js app
npx create-next-app@latest fromfarmers --typescript --tailwind --eslint --app --src-dir --import-alias "@/*"
```

```bash

npx prisma init
```

### Phase 3: Database Schema Adaptation

#### Step 3.1: Core Models for FromFarmers

```prisma
// prisma/schema.prisma

generator client {
  provider = "prisma-client-js"
}

datasource db {
  provider = "postgresql"
  url      = env("DATABASE_URL")
}

// ============ ENUMS ============
enum UserRole {
  BUYER
  FARMER
  ADMIN
}

enum ApprovalStatus {
  PENDING
  APPROVED
  REJECTED
}

enum OrderStatus {
  PENDING
  CONFIRMED
  READY_FOR_PICKUP
  OUT_FOR_DELIVERY
  DELIVERED
  CANCELLED
}

enum ProductCategory {
  VEGETABLES
  FRUITS
  GRAINS
  DAIRY
  MEAT
  POULTRY
  EGGS
  HONEY
  HERBS
  ORGANIC
  PRESERVED
  BAKED_GOODS
  BEVERAGES
  OTHER
}

enum DeliveryMethod {
  PICKUP
  DELIVERY
  BOTH
}

enum UnitType {
  KG
  GRAM
  PIECE
  BUNCH
  DOZEN
  LITER
  BOX
  BAG
}

// ============ MODELS ============

model User {
  id                   String         @id @default(uuid())
  email                String         @unique
  password             String
  firstName            String
  lastName             String
  role                 UserRole       @default(BUYER)
  phone                String?
  profileImage         String?
  emailVerified        Boolean        @default(false)
  verificationToken    String?        @unique
  createdAt            DateTime       @default(now())
  updatedAt            DateTime       @updatedAt
  
  // Relations
  farmer               Farmer?
  orders               Order[]        @relation("BuyerOrders")
  notifications        Notification[]
  reviews              Review[]
  favorites            Favorite[]

  @@index([email])
}

model Farmer {
  id                   String         @id @default(uuid())
  userId               String         @unique
  user                 User           @relation(fields: [userId], references: [id])
  
  // Farm Details
  farmName             String
  slug                 String?        @unique
  description          String?
  farmImages           String[]
  profileImage         String?
  backgroundImage      String?
  
  // Location
  province             String
  city                 String
  farmAddress          String?
  latitude             Float?
  longitude            Float?
  
  // Contact
  contactPhone         String?
  whatsapp             String?
  website              String?
  
  // Farming Details
  farmType             String[]       // organic, conventional, hydroponic, etc.
  certifications       String[]       // organic certified, farm-assured, etc.
  yearsOperating       Int?
  farmSize             String?        // e.g., "5 hectares"
  
  // Delivery Options
  deliveryMethod       DeliveryMethod @default(PICKUP)
  deliveryRadius       Int?           // km radius for delivery
  deliveryFee          Float?
  pickupLocation       String?        // if different from farm address
  pickupHours          Json?          // operating hours for pickup
  
  // Status
  approvalStatus       ApprovalStatus @default(PENDING)
  isVerified           Boolean        @default(false)
  avgRating            Float          @default(0)
  totalSales           Int            @default(0)
  
  // Relations
  products             FarmProduct[]
  orders               Order[]        @relation("FarmerOrders")
  reviews              Review[]
  
  createdAt            DateTime       @default(now())
  updatedAt            DateTime       @updatedAt

  @@index([approvalStatus])
  @@index([province])
  @@index([city])
}

model FarmProduct {
  id                   String          @id @default(uuid())
  farmerId             String
  farmer               Farmer          @relation(fields: [farmerId], references: [id])
  
  // Product Details
  name                 String
  slug                 String?         @unique
  description          String
  images               String[]
  category             ProductCategory
  
  // Pricing
  price                Float
  unit                 UnitType
  minOrder             Int             @default(1)
  maxOrder             Int?
  isOnSale             Boolean         @default(false)
  salePrice            Float?
  
  // Stock & Availability
  stock                Int             @default(0)
  isAvailable          Boolean         @default(true)
  harvestDate          DateTime?       // when was it harvested
  expiryDate           DateTime?       // sell by date
  
  // Farming Details
  isOrganic            Boolean         @default(false)
  farmingMethod        String?         // e.g., "free-range", "grass-fed"
  storageInstructions  String?
  
  // Purchase Options
  whatsappNumber       String?
  websiteUrl           String?
  purchaseNote         String?
  
  // Status
  approvalStatus       ApprovalStatus  @default(PENDING)
  viewCount            Int             @default(0)
  
  // Relations
  orders               OrderItem[]
  favorites            Favorite[]
  
  createdAt            DateTime        @default(now())
  updatedAt            DateTime        @updatedAt

  @@index([farmerId])
  @@index([category])
  @@index([approvalStatus])
  @@index([slug])
}

model Order {
  id                   String          @id @default(uuid())
  buyerId              String
  buyer                User            @relation("BuyerOrders", fields: [buyerId], references: [id])
  farmerId             String
  farmer               Farmer          @relation("FarmerOrders", fields: [farmerId], references: [id])
  
  // Order Details
  items                OrderItem[]
  totalAmount          Float
  status               OrderStatus     @default(PENDING)
  
  // Delivery/Pickup
  deliveryMethod       DeliveryMethod
  deliveryAddress      String?
  deliveryDate         DateTime?
  deliveryNotes        String?
  
  // Contact
  buyerPhone           String?
  
  createdAt            DateTime        @default(now())
  updatedAt            DateTime        @updatedAt

  @@index([buyerId])
  @@index([farmerId])
  @@index([status])
}

model OrderItem {
  id                   String          @id @default(uuid())
  orderId              String
  order                Order           @relation(fields: [orderId], references: [id])
  productId            String
  product              FarmProduct     @relation(fields: [productId], references: [id])
  
  quantity             Int
  unitPrice            Float
  totalPrice           Float

  @@index([orderId])
  @@index([productId])
}

model Review {
  id                   String          @id @default(uuid())
  rating               Int             // 1-5
  comment              String?
  userId               String
  user                 User            @relation(fields: [userId], references: [id])
  farmerId             String
  farmer               Farmer          @relation(fields: [farmerId], references: [id])
  
  // Response
  farmerResponse       String?
  farmerRespondedAt    DateTime?
  
  approvalStatus       ApprovalStatus  @default(PENDING)
  createdAt            DateTime        @default(now())
  updatedAt            DateTime        @updatedAt

  @@index([farmerId])
  @@index([userId])
}

model Favorite {
  userId               String
  user                 User            @relation(fields: [userId], references: [id])
  productId            String
  product              FarmProduct     @relation(fields: [productId], references: [id])
  createdAt            DateTime        @default(now())

  @@id([userId, productId])
  @@index([productId])
}

model Notification {
  id                   String          @id @default(uuid())
  userId               String
  user                 User            @relation(fields: [userId], references: [id])
  message              String
  link                 String?
  isRead               Boolean         @default(false)
  createdAt            DateTime        @default(now())

  @@index([userId])
}
```
